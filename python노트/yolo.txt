Y = w*x + b
15 = w*2 + b

(정답)
w = 6
b = 3

x = 2
y = 15

# 계산을 이렇게 : 15 = w * 2 + b
# 영향도 계산 : 편미분 한다.
# 관련 값외의 값은 상수 취급
# y-pred 에 의해 연결되어 있다. 	# 연쇄법칙

L = (y-pred - y)^2	// 제곱   	# 손실함수
y-pred =  w*2 + b 		    	# 예측함수
dL/dw : 연쇄법칙(Chain Rule)
grad_w = dL/dy-pred * dy-pred/dw
grad_b = dL/dy-pred * dy-pred/db 

w -= lr * grad_w 				# learning rate 학습률,  lr = 0.001 정해진 값
b -= lr * grad_b 

y', y-pred : 신경망이 계산한 값
y , y-true : 실제값
